cmake_minimum_required(VERSION 3.16)
project(calendar_ui_preview LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(CALENDAR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(LVGL_SOURCE_DIR "${CALENDAR_ROOT}/.pio/libdeps/rgb-staged-startup/lvgl" CACHE PATH "Resolved LVGL 9.2.2 checkout")
file(READ "${LVGL_SOURCE_DIR}/library.json" LVGL_MANIFEST)
if(NOT LVGL_MANIFEST MATCHES "\"version\": *\"9.2.2\"")
    message(FATAL_ERROR "This preview requires the project's resolved LVGL 9.2.2")
endif()
set(LV_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/preview_lv_conf.h" CACHE STRING "Preview-only configuration" FORCE)
set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/lvgl")
file(GLOB GENERATED_SOURCES "${CALENDAR_ROOT}/src/ui_generated/*.c")
set(CORE_SOURCES calendar_count_format calendar_types calendar_provider calendar_period calendar_navigation ical_feed theme_palette weather_parser)
foreach(NAME IN LISTS CORE_SOURCES)
    list(APPEND APP_SOURCES "${CALENDAR_ROOT}/src/app/${NAME}.cpp")
endforeach()
if(EXISTS "${CALENDAR_ROOT}/src/app/ui_status.cpp")
    list(APPEND APP_SOURCES "${CALENDAR_ROOT}/src/app/ui_status.cpp")
endif()
add_executable(calendar_ui_preview preview.cpp services.cpp ${GENERATED_SOURCES} ${APP_SOURCES}
    "${CALENDAR_ROOT}/src/app/ui_controller.cpp"
    "${CALENDAR_ROOT}/src/app/ui_actions.cpp"
    "${CALENDAR_ROOT}/src/app/weather_icon.cpp"
    "${CALENDAR_ROOT}/src/app/weather_icon_assets.cpp")
target_include_directories(calendar_ui_preview PRIVATE stubs "${CALENDAR_ROOT}/include" "${CALENDAR_ROOT}/src")
target_compile_definitions(calendar_ui_preview PRIVATE APP_RGB_STAGED_STARTUP=1)
target_link_libraries(calendar_ui_preview PRIVATE lvgl m)
target_link_options(calendar_ui_preview PRIVATE "-Wl,--wrap=time")
